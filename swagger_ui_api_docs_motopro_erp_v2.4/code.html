Integración Frontend / PWA

Cliente JavaScript Universal (SDK Boilerplate)

Ready for Browser & Electron 
// 1. Inicialización de configuración global en Shell o Window
window.MOTOPRO_CONFIG = {
  apiBaseUrl: 'https://api.tu-taller.com/v1',
  activeBranchId: 'SEDE-NORTE-BOG',
  hardwareEscPosEndpoint: 'http://localhost:9100/escpos',
  syncIntervalMs: 5000
};

// 2. MotoPro HTTP Client con auto-inyectado de Token y Sede
class MotoProApiClient {
  static async request(endpoint, options = {}) {
    const token = localStorage.getItem('motopro_jwt');
    const headers = {
      'Content-Type': 'application/json',
      'X-Workshop-Branch-Id': window.MOTOPRO_CONFIG.activeBranchId,
      ...(token && { 'Authorization': `Bearer ${token}` }),
      ...options.headers
    };
    return fetch(`${window.MOTOPRO_CONFIG.apiBaseUrl}${endpoint}`, { ...options, headers })
      .then(res => res.json());
  }
}
